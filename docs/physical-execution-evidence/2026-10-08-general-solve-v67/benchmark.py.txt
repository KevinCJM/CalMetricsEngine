@njit(nogil=True)
def direct_reference(h,g,z,b,zr):
    # One private final coefficient allocation, directly filled in F order.
    # np.linalg.solve is the same mature partial-pivot LAPACK computation.
    n=h.shape[0];m=g.shape[0];size=n+m
    coefficient=np.empty((size,size)).T
    if h.flags.c_contiguous:
        # Tile the layout conversion; avoid a full strided column sweep.
        for j0 in range(0,n,64):
            for i0 in range(0,n,64):
                for i in range(i0,min(i0+64,n)):
                    for j in range(j0,min(j0+64,n)):coefficient[i,j]=h[i,j]
    else:
        for j in range(n):
            for i in range(n):coefficient[i,j]=h[i,j]
    for j in range(n):
        for i in range(m):coefficient[n+i,j]=g[i,j]
    for j in range(m):
        for i in range(n):coefficient[i,n+j]=g[j,i]
        for i in range(m):coefficient[n+i,n+j]=z[i,j]
    rhs=np.empty((size,1))
    for i in range(n):rhs[i,0]=b[i,0]
    for i in range(m):rhs[n+i,0]=zr[i,0]
    return np.linalg.solve(coefficient,rhs)

@njit(nogil=True)
def direct_c_reference(h,g,z,b,zr):
    n=h.shape[0];m=g.shape[0];size=n+m
    coefficient=np.empty((size,size))
    for i in range(n):
        for j in range(n):coefficient[i,j]=h[i,j]
        for j in range(m):coefficient[i,n+j]=g[j,i]
    for i in range(m):
        for j in range(n):coefficient[n+i,j]=g[i,j]
        for j in range(m):coefficient[n+i,n+j]=z[i,j]
    rhs=np.empty((size,1))
    for i in range(n):rhs[i,0]=b[i,0]
    for i in range(m):rhs[n+i,0]=zr[i,0]
    return np.linalg.solve(coefficient,rhs)

@njit(nogil=True)
def concat_reference(h,g,z,b,zr):
    top=np.concatenate((h,g.T),axis=1)
    bottom=np.concatenate((g,z),axis=1)
    coefficient=np.concatenate((top,bottom),axis=0)
    rhs=np.concatenate((b,zr),axis=0)
    return np.linalg.solve(coefficient,rhs)
